# Bear Den TV — Qt 6 / Qt Quick TV shell.
#
# Build from the repository root with the pinned toolchain sourced (`. scripts/env.sh`):
#   cmake -S apps/tv-shell -B build/tv-shell -G Ninja && cmake --build build/tv-shell
#   cd build/tv-shell && QT_QPA_PLATFORM=offscreen ctest --output-on-failure
#
# Layout: the whole shell (C++ host types + QML + assets) is one static QML module `BearDen`
# backed by the `beardenshell` library; the executable and every test link the same module,
# so tests exercise the shipped QML rather than copies.
cmake_minimum_required(VERSION 3.28)
project(BearDenTvShell VERSION 0.1.0 LANGUAGES CXX)

option(BDTV_BUILD_TESTS "Build the offscreen ctest suite" ON)

include(cmake/ToolchainOpenGL.cmake)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network Svg QuickTest Test)
qt_standard_project_setup(REQUIRES 6.8)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra)

# Pixel art generated by tools/pixelart (bears, the mark), the Classic art
# (SVG: bears and the mark in assets/, assets/classic/), and the built-in
# ornaments (a PNG for Pixel and an SVG for Classic each): globbed so
# regenerating the art needs no list edits.
file(GLOB BDTV_PIXEL_ASSETS CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}
  assets/*.svg assets/classic/*.svg assets/classic/*.png assets/pixel/*.png assets/ornaments/*.png assets/ornaments/*.svg)
set(BDTV_ASSETS ${BDTV_PIXEL_ASSETS})
# QML files live under qml/ on disk but are addressed as module roots (qrc:/qt/qml/BearDen/X.qml).
foreach(f IN LISTS BDTV_QML_FILES)
  get_filename_component(alias "${f}" NAME)
  set_source_files_properties("${f}" PROPERTIES QT_RESOURCE_ALIAS "${alias}")
endforeach()
set_source_files_properties(qml/Apps.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
set_source_files_properties(qml/World.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)

qt_add_library(beardenshell STATIC)
qt_add_qml_module(beardenshell
  URI BearDen
  VERSION 1.0
  QML_FILES ${BDTV_QML_FILES}
  RESOURCES ${BDTV_ASSETS}
  SOURCES
    src/Theme.h src/Theme.cpp
    src/ThemeRegistry.h src/ThemeRegistry.cpp
    src/FocusMemory.h src/FocusMemory.cpp
    src/ItemsModel.h src/ItemsModel.cpp
    src/SectionsModel.h src/SectionsModel.cpp
    src/SessionModel.h src/SessionModel.cpp
    src/Navigator.h src/Navigator.cpp
    src/IpcClient.h src/IpcClient.cpp
    src/QrRenderer.h src/QrRenderer.cpp
    src/RoundedImage.h src/RoundedImage.cpp
    src/ShellController.h src/ShellController.cpp
)
# Built-in theme packages (the repository's themes/<id>/), compiled in at :/themes/<id>/.
# Owners add themes at runtime in $XDG_DATA_HOME/bear-den-tv/themes/<id>/ (docs/THEMES.md).
set(BDTV_THEMES_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../themes")
file(GLOB_RECURSE BDTV_THEME_FILES CONFIGURE_DEPENDS
  "${BDTV_THEMES_DIR}/*.json" "${BDTV_THEMES_DIR}/*.svg" "${BDTV_THEMES_DIR}/*.png"
  "${BDTV_THEMES_DIR}/*.jpg" "${BDTV_THEMES_DIR}/*.jpeg" "${BDTV_THEMES_DIR}/*.webp")
qt_add_resources(beardenshell "bdtv_themes" PREFIX "/themes" BASE "${BDTV_THEMES_DIR}" FILES ${BDTV_THEME_FILES})
target_include_directories(beardenshell PUBLIC src)
target_link_libraries(beardenshell PUBLIC Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Svg)

# --- Executable --------------------------------------------------------------------------
qt_add_executable(bear-den-tv-shell src/main.cpp)
target_link_libraries(bear-den-tv-shell PRIVATE beardenshell beardenshellplugin)
